// Inicialización de Culqi Checkout function initCulqi() { if (typeof Culqi !== 'undefined') { Culqi.publicKey = CULQI_PUBLIC_KEY; Culqi.options({ lang: 'auto', installments: false, paymentMethods: { tarjeta: false, yape: true, billetera: false, bancaMovil: false, agente: false, cuotealo: false } }); } } // Flujo corregido con generación de orden obligatoria para Yape document.getElementById('pay-yape-btn').addEventListener('click', async () => { const payBtn = document.getElementById('pay-yape-btn'); if (typeof Culqi === 'undefined') { showToast("Cargando pasarela segura de Yape...", "⏳"); return; } // Deshabilitar botón temporalmente para evitar doble clic payBtn.disabled = true; payBtn.innerText = "Conectando con Yape..."; showToast("Generando orden segura...", "⏳"); try { // 1. Crear la orden oficial en el backend const clientEmail = currentCalculation?.worker?.email || "cliente@liquidaya.pe"; const res = await fetch(`${BACKEND_URL}/api/create-order`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ amount: 990, currency: 'PEN', description: 'Expediente Liquidación Laboral LiquidaYa', email: clientEmail }) }); const data = await res.json(); if (data.status !== "ok" || !data.order_id) { showToast("No se pudo iniciar la orden: " + (data.message || "Error de conexión"), "⚠️"); payBtn.disabled = false; payBtn.innerText = "Pagar S/ 9.90 con Yape"; return; } // 2. Abrir Culqi vinculando el order_id obligatorio initCulqi(); Culqi.settings({ title: 'LiquidaYa Perú', currency: 'PEN', amount: 990, order: data.order_id // <-- ESTO RESUELVE EL ERROR CCKT-408 }); Culqi.open(); } catch (err) { showToast("Error conectando con el servidor. Reintenta.", "⚠️"); } finally { payBtn.disabled = false; payBtn.innerText = "Pagar S/ 9.90 con Yape"; } }); // Callback de confirmación de pago de Culqi window.culqi = async function() { // Con órdenes de Yape, Culqi responde mediante Culqi.order if (Culqi.order) { Culqi.close(); showToast("¡Pago con Yape procesado con éxito!", "✅"); unlockExpediente(); } else if (Culqi.token) { Culqi.close(); showToast("Pago validado.", "✅"); unlockExpediente(); } else if (Culqi.error) { showToast(Culqi.error.user_message || "No se completó la operación.", "⚠️"); } }; window.addEventListener('load', initCulqi);
